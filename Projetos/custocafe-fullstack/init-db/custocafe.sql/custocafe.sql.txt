-- ============================================
-- 1. Tabelas
-- ============================================

CREATE TABLE IF NOT EXISTS usuarios (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    senha VARCHAR(255) NOT NULL,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS setores (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao TEXT
);

CREATE TABLE IF NOT EXISTS glebas (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    setor_id INT REFERENCES setores(id) ON DELETE SET NULL,
    area_hectares DECIMAL(10,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS mao_obra_contratada (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    funcao VARCHAR(100),
    custo_hora DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS insumos (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    tipo VARCHAR(50),
    preco_unitario DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS maquinas_automotoras (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    custo_hora DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS maquinas_implementos (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    custo_hora DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS benfeitorias (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    custo DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS veiculos (
    id SERIAL PRIMARY KEY,
    modelo VARCHAR(100) NOT NULL,
    custo_hora DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS operacoes_servicos (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    custo DECIMAL(12,2) DEFAULT 0
);

CREATE TABLE IF NOT EXISTS aplicacao_adubos (
    id SERIAL PRIMARY KEY,
    gleba_id INT REFERENCES glebas(id) ON DELETE CASCADE,
    insumo_id INT REFERENCES insumos(id) ON DELETE CASCADE,
    quantidade DECIMAL(12,2) DEFAULT 0,
    custo DECIMAL(12,2) DEFAULT 0,
    data_aplicacao DATE DEFAULT CURRENT_DATE
);

CREATE TABLE IF NOT EXISTS aplicacao_defensivos (
    id SERIAL PRIMARY KEY,
    gleba_id INT REFERENCES glebas(id) ON DELETE CASCADE,
    insumo_id INT REFERENCES insumos(id) ON DELETE CASCADE,
    quantidade DECIMAL(12,2) DEFAULT 0,
    custo DECIMAL(12,2) DEFAULT 0,
    data_aplicacao DATE DEFAULT CURRENT_DATE
);

CREATE TABLE IF NOT EXISTS aplicacao_herbicidas (
    id SERIAL PRIMARY KEY,
    gleba_id INT REFERENCES glebas(id) ON DELETE CASCADE,
    insumo_id INT REFERENCES insumos(id) ON DELETE CASCADE,
    quantidade DECIMAL(12,2) DEFAULT 0,
    custo DECIMAL(12,2) DEFAULT 0,
    data_aplicacao DATE DEFAULT CURRENT_DATE
);

CREATE TABLE IF NOT EXISTS producao (
    id SERIAL PRIMARY KEY,
    gleba_id INT REFERENCES glebas(id) ON DELETE CASCADE,
    cultura VARCHAR(100) NOT NULL,
    quantidade DECIMAL(12,2) DEFAULT 0,
    preco_unitario DECIMAL(12,2) DEFAULT 0,
    data_producao DATE DEFAULT CURRENT_DATE
);

CREATE TABLE IF NOT EXISTS centro_custos (
    id SERIAL PRIMARY KEY,
    descricao VARCHAR(255) NOT NULL,
    valor DECIMAL(12,2) DEFAULT 0,
    mao_obra_id INT REFERENCES mao_obra_contratada(id),
    maquina_id INT REFERENCES maquinas_automotoras(id),
    data_lancamento DATE DEFAULT CURRENT_DATE
);

CREATE TABLE IF NOT EXISTS ordem_servicos (
    id SERIAL PRIMARY KEY,
    descricao VARCHAR(255) NOT NULL,
    valor DECIMAL(12,2) DEFAULT 0,
    gleba_id INT REFERENCES glebas(id) ON DELETE CASCADE,
    data_servico DATE DEFAULT CURRENT_DATE
);

-- Índices
CREATE INDEX IF NOT EXISTS idx_gleba_setor ON glebas(setor_id);
CREATE INDEX IF NOT EXISTS idx_aplicacao_adubos_gleba ON aplicacao_adubos(gleba_id);
CREATE INDEX IF NOT EXISTS idx_aplicacao_defensivos_gleba ON aplicacao_defensivos(gleba_id);
CREATE INDEX IF NOT EXISTS idx_aplicacao_herbicidas_gleba ON aplicacao_herbicidas(gleba_id);
CREATE INDEX IF NOT EXISTS idx_producao_gleba ON producao(gleba_id);
CREATE INDEX IF NOT EXISTS idx_ordem_servicos_gleba ON ordem_servicos(gleba_id);
